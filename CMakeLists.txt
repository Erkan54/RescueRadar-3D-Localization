cmake_minimum_required(VERSION 3.16)
project(RescueRadarSamples LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(rssi_filter src/KalmanRSSIFilter.cpp)
target_include_directories(rssi_filter PUBLIC src)
add_executable(rssi_demo examples/esp32_rssi_smoothing.cpp)
target_link_libraries(rssi_demo PRIVATE rssi_filter)
include(CTest)
if(BUILD_TESTING)
    add_executable(kalman_test tests/kalman_test.cpp)
    target_link_libraries(kalman_test PRIVATE rssi_filter)
    add_test(NAME kalman_behavior COMMAND kalman_test)
endif()
